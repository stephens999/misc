Introduction

After two “quick and dirty” attempts, I think I’m now ready to implement a proper version of an asymmetric version of fastica. We can also think of this as a Newton-based method for optimizing the Empirical Bayes Projection model (EBProj).

The Newton version of EBCD

Given a psd matrix \(S=UDU'\) and a residual variance \(\sigma^2\), define the objective function \(J\) as: \[J(w, g; \sigma^2, S) = \sum_{i=1}^n \log Z((UDw)_i, g, \sigma^2)\] where \[Z(x, g, \sigma^2) = \int g(\theta) \exp\left( \frac{2x \theta - \theta^2}{2\sigma^2} \right) d\theta. \] The EBCD method (for fixed \(\sigma^2\)) can be implemented by maximizing \(J\) subject to \(w'Dw=1\) using a Newton method on \(w\) (and, for example, EM method for \(g\)). Given the optimal \(w,g\), the optimal \(q\) is the EBNM posterior with prior \(g^*\) and data \(x=UDw^*,s=\sigma\).

For \(g\) a binary prior with probabilities \(1-p,p\) on \(y_0,y_1\) we have \[Z(x, g, \sigma^2) = (1-p)\exp\!\left(\frac{2xy_0 - y_0^2}{2\sigma^2}\right) + p\exp\!\left(\frac{2xy_1 - y_1^2}{2\sigma^2}\right).\]

Define the log unnormalized weights \(\ell_j(x) = \log p_j + (2xy_j - y_j^2)/(2\sigma^2)\), with \(p_0=1-p\) and \(p_1=p\). Then the posterior probability of \(y_1\) is \[\pi_1(x) = \frac{e^{\ell_1(x)}}{e^{\ell_0(x)} + e^{\ell_1(x)}},\] the posterior mean is \[M(x;\sigma^2,g) = (1-\pi_1)y_0 + \pi_1 y_1,\] and the derivative of \(M\) with respect to \(x\) (used in the Newton step) is \[M'(x;\sigma^2,g) = \pi_1(1-\pi_1)(y_1-y_0)^2/\sigma^2.\]

The EBProj approach

The EBProj method works on a projection matrix \(P\) (if \(S\) is a whitened then \(P=S/n\) is a projection matrix). The objective is: \[F_{\text{EBproj}}(w, g; \tau, P) = \frac{n - k - 2b}{2} \left( \log \tau - \tau + 1 \right) + J(w,g; \sigma^2_\text{eff}, S = nP)\] where \(\sigma^2_\text{eff} =\frac{n}{(n - k - 2b)\tau}\).

Optimizing the EBProj objective can iterate between optimizing \(J\) over \(w,g\) as above and then optimizing \(\tau = 1/(1-\hat\rho)\) where \(\hat\rho = \bar{\mathbf{v}}'P\bar{\mathbf{v}} / U_q\), \(U_q = E_q[\mathbf{v}'\mathbf{v}]\) is the second moment, and \(P=S/n\).

Implementation

Helper functions for the binary prior

# log Z(x, g, sigma2) for binary prior g = (1-p) delta_{y0} + p delta_{y1}
log_Z_binary = function(x, y_0, y_1, p, sigma2) {
  lp0 = log(1 - p) + (2*x*y_0 - y_0^2) / (2*sigma2)
  lp1 = log(p)     + (2*x*y_1 - y_1^2) / (2*sigma2)
  m   = pmax(lp0, lp1)
  log(exp(lp0 - m) + exp(lp1 - m)) + m
}

# posterior probability of y_1
post_prob1 = function(x, y_0, y_1, p, sigma2) {
  lp0 = log(1 - p) + (2*x*y_0 - y_0^2) / (2*sigma2)
  lp1 = log(p)     + (2*x*y_1 - y_1^2) / (2*sigma2)
  m   = pmax(lp0, lp1)
  e0  = exp(lp0 - m);  e1 = exp(lp1 - m)
  e1 / (e0 + e1)
}

# posterior mean M(x; sigma2, g)
post_mean_binary = function(x, y_0, y_1, p, sigma2) {
  pi1 = post_prob1(x, y_0, y_1, p, sigma2)
  (1 - pi1)*y_0 + pi1*y_1
}

# dM/dx = pi1*(1-pi1)*(y1-y0)^2 / sigma2
post_mean_deriv_binary = function(x, y_0, y_1, p, sigma2) {
  pi1 = post_prob1(x, y_0, y_1, p, sigma2)
  pi1 * (1 - pi1) * (y_1 - y_0)^2 / sigma2
}

Initialisation

Define a fit object (simply a list) that contains the data \(U\), and the parameters to be updated (prior \(g\), weights \(w\)). Subsequent update functions will take a fit as input and output a new fit.

ebproj_init = function(U, b=0, c = 1, tau=NULL, sigma2=NULL, w=NULL) {
  n   = nrow(U)
  k   = ncol(U)
  nu  = n - k - 2*b
  if (!is.null(tau) && !is.null(sigma2))
    stop("specify at most one of tau and sigma2")
  if (is.null(tau) && is.null(sigma2)) tau = 1
  if (is.null(sigma2)) sigma2 = n / (nu * tau)
  if (is.null(tau))    tau    = n / (nu * sigma2)
  d   = rep(n, k)   # S = UDU' = nUU' = nP
  
  if(is.null(w)){
    w   = rnorm(k)
    w   = w / sqrt(sum(d * w^2))   # normalise to w'Dw = 1
  }
  
  fit = list(
    U         = U,
    d         = d,
    b         = b,
    n         = n,
    k         = k,
    nu        = nu,
    sigma2    = sigma2,
    w         = w,
    c         = c,     # scale factor; effective atoms are c*y_0, c*y_1
    y_0       = -1,
    y_1       = 1,
    p         = 0.5,
    objective = NULL,
    tau       = tau
  )
  return(fit)
}

Objective function

ebproj_objective = function(fit) {
  ey0 = fit$c * fit$y_0;  ey1 = fit$c * fit$y_1   # effective atoms
  x   = ebproj_calc_x(fit)
  J   = sum(log_Z_binary(x, ey0, ey1, fit$p, fit$sigma2))
  obj = (fit$nu/2) * (log(fit$tau) - fit$tau + 1) + J
  return(obj)
}

The dual objective (eq 200 of the notes) is \(F_{\text{EBproj}}(\mathbf{w}, g; \tau) = J(\mathbf{w}, g; \sigma^2_\text{eff}) + C(\tau)\). It is equivalent to the primal at the optimal \(q\) but is easier to compute because it avoids computing \(\text{KL}(q\|g)\) explicitly.

The primal objective (eq 190/199) retains the full variational structure: \[F_{\text{EBproj}}(q, g; \tau) = \frac{n - k - 2b}{2} \bigl(\log \tau - \tau(1 - \hat\rho) + 1\bigr) - \text{KL}(q \| g)\] where \(\hat\rho = \bar{\mathbf{v}}'\mathbf{P}\bar{\mathbf{v}} / U_q\), \(\bar{\mathbf{v}} = E_q[\mathbf{v}]\) is the posterior mean, and \(U_q = E_q[\mathbf{v}'\mathbf{v}]\) is the second moment.

For the binary prior, \(\text{KL}(q\|g) = \sum_i \bigl[(1-\pi_{1i})\log\tfrac{1-\pi_{1i}}{1-p} + \pi_{1i}\log\tfrac{\pi_{1i}}{p}\bigr]\) and \(U_q = \sum_i \bigl[(1-\pi_{1i})(cy_0)^2 + \pi_{1i}(cy_1)^2\bigr]\).

kl_binary = function(pi1, p) {
  pi0 = 1 - pi1
  # clamp to avoid log(0)
  pi0 = pmax(pi0, 1e-300);  pi1 = pmax(pi1, 1e-300)
  sum(pi0 * log(pi0 / (1 - p)) + pi1 * log(pi1 / p))
}

ebproj_primal_objective = function(fit) {
  ey0 = fit$c * fit$y_0;  ey1 = fit$c * fit$y_1
  x    = ebproj_calc_x(fit)
  pi1  = post_prob1(x, ey0, ey1, fit$p, fit$sigma2)
  vbar = post_mean_binary(x, ey0, ey1, fit$p, fit$sigma2)

  Uv   = as.vector(t(fit$U) %*% vbar)
  Uq   = sum((1 - pi1) * ey0^2 + pi1 * ey1^2)   # E_q[v'v] second moment
  # rho_hat = vbar' P vbar / U_q, with P = S/n and S = UDU'
  rho  = sum(fit$d * Uv^2) / (fit$n * Uq)

  kl   = kl_binary(pi1, fit$p)
  obj  = (fit$nu/2) * (log(fit$tau) - fit$tau * (1 - rho) + 1) - kl
  return(obj)
}

Prior parameterisation and update modes

The binary prior is parameterised as \((c, g_0)\) where \(g_0 = (1-p)\delta_{y_0} + p\delta_{y_1}\) is a base prior and \(c > 0\) is a scale factor. The effective prior is \[g = (1-p)\delta_{cy_0} + p\delta_{cy_1},\] so the effective atoms are \(cy_0\) and \(cy_1\). This parameterisation is over-complete — multiplying \(c\) by any positive constant and dividing \(y_0, y_1\) by the same constant leaves \(g\) unchanged — but the redundancy is useful because it separates “shape” from “scale”.

A natural constraint is to require \(g_0\) to have mean 0 and unit variance: \[\mathbb{E}_{g_0}[v] = 0 \quad \text{and} \quad \text{Var}_{g_0}(v) = 1.\] Under these constraints \(g_0\) has a single free parameter \(p\), with atoms determined by \[y_1 = \sqrt{\frac{1-p}{p}}, \qquad y_0 = -\sqrt{\frac{p}{1-p}}.\] The scale \(c\) then controls the overall amplitude of \(g\).

Three update functions are provided, selected via the g_update argument of ebproj_fit:

g_update function called what is updated
"g" ebproj_update_g full EM: \((p, y_0, y_1)\), rescale \(g_0\) unit variance
"c" ebproj_update_c scale \(c\) only; \(g_0 = (p, y_0, y_1)\) fixed
"g0" ebproj_update_g0 constrained \(g_0\): optimise \(p\) with \(y_0, y_1\) set by mean-0, var-1 constraint; \(c\) fixed
"none" — freeze \(g\) entirely

ebproj_update_g (full EM): updates \(p\), \(y_0\), \(y_1\) jointly via their EM closed forms, then rescales so that \(\text{Var}(g_0) = 1\), storing the scale in \(c\).

ebproj_update_c: fixes \(g_0 = (p, y_0, y_1)\) and finds \(c\) by exact 1-D maximisation of \(J(w, cg_0)\) over \(c > 0\) on the log scale.

ebproj_update_g0: fixes \(c\) and optimises \(p \in (0,1)\) subject to the mean-0, unit-variance constraint, via 1-D numerical maximisation. After updating \(p\) the atoms are recomputed as \(y_1 = \sqrt{(1-p)/p}\), \(y_0 = -\sqrt{p/(1-p)}\).

Note that only the last of these constrains \(g_0\) to have zero mean, so that update is a bit inconsistent with the others. For example, if you estimate \(g\) freely, and then perfom ebproj_update_g0 the solution will likely change to satisfy the constraint.

Update functions

Each function takes a fit object and returns a fit object.

The Newton update for \(w\) (Section 2.3 of the notes) solves the constrained problem with the approximate Newton step \[w_{\text{new}} = (\lambda I_k - \tilde{H})^{-1}(w_\text{target} - \tilde{H}w)\] where \(w_\text{target} = U^T M(x)\) and \(\lambda = \max(w^T D w_\text{target},\, \max_j \tilde{H}_{jj} + \epsilon)\). Five choices of diagonal preconditioner \(\tilde{H}\) are supported (default "ica"):

  1. "diag" (Full Diagonal): \(\tilde{H}_{jj} = c_j d_j\) where \(c_j = \sum_i U_{ij}^2 M'(x_i)\).
  2. "trace" (Trace): \(\tilde{H}_{jj} = c_\text{trace}\, d_j\) where \(c_\text{trace} = \sum_i M'(x_i) S_{ii} / \operatorname{tr}(D)\) and \(S_{ii} = \sum_j U_{ij}^2 d_j\).
  3. "iso" (Isotropic): \(\tilde{H}_{jj} = c_\text{iso}\) where \(c_\text{iso} = (1/k)\sum_i M'(x_i) S_{ii}\).
  4. "ica" (FastICA): \(\tilde{H}_{jj} = c_\text{ica}\) where \(c_\text{ica} = \sum_i M'(x_i)\) — the isotropic approximation under the uniform-feature-variance assumption \(S_{ii} \approx k\) (Lemma 2.1).
  5. "none" (\(\tilde{H} = 0\)): recovers the natural power / gradient step \(w_\text{new} \propto w_\text{target}\).
ebproj_update_w = function(fit, hessian = "ica", eps = 1e-6) {
  ey0 = fit$c * fit$y_0;  ey1 = fit$c * fit$y_1
  x        = ebproj_calc_x(fit)          # UDw, length n
  Mx       = post_mean_binary(x, ey0, ey1, fit$p, fit$sigma2)
  Mpx      = post_mean_deriv_binary(x, ey0, ey1, fit$p, fit$sigma2)

  w_target = as.vector(t(fit$U) %*% Mx)                    # U^T M(x), length k

  # diagonal of S: S_ii = sum_j U_ij^2 d_j
  S_diag = as.vector((fit$U^2) %*% fit$d)                  # length n

  H_diag = switch(hessian,
    diag  = { c_j = as.vector(t(fit$U^2) %*% Mpx); c_j * fit$d },
    trace = { c_t = sum(Mpx * S_diag) / sum(fit$d); c_t * fit$d },
    iso   = { c_i = sum(Mpx * S_diag) / fit$k; rep(c_i, fit$k) },
    ica   = { c_c = sum(Mpx); rep(c_c, fit$k) },
    none  = rep(0, fit$k),
    stop("hessian must be one of: 'diag', 'trace', 'iso', 'ica', 'none'")
  )

  # Lagrange multiplier; ensure lambda > max(H_diag) for positive definiteness
  lambda = max(sum(fit$d * fit$w * w_target), max(H_diag) + eps)

  w_new  = (w_target - H_diag * fit$w) / (lambda - H_diag)
  # normalise to w'Dw = 1
  scale  = sqrt(sum(fit$d * w_new^2))
  fit$w  = w_new / scale
  return(fit)
}

# Full EM update for binary prior: update (p, y_0, y_1), then rescale so that
# Var(g0) = 1, absorbing the scale into c.
ebproj_update_g = function(fit) {
  ey0 = fit$c * fit$y_0;  ey1 = fit$c * fit$y_1
  x   = ebproj_calc_x(fit)
  pi1 = post_prob1(x, ey0, ey1, fit$p, fit$sigma2)
  pi0 = 1 - pi1

  fit$p  = mean(pi1)
  y0_new = sum(pi0 * x) / sum(pi0)
  y1_new = sum(pi1 * x) / sum(pi1)

  # rescale so Var(g0) = 1; c carries the scale
  var_g  = (1 - fit$p) * y0_new^2 + fit$p * y1_new^2 -
           ((1 - fit$p) * y0_new + fit$p * y1_new)^2
  fit$c   = sqrt(var_g)
  fit$y_0 = y0_new / fit$c
  fit$y_1 = y1_new / fit$c
  return(fit)
}

# Scale-only update: fix the base prior g_0 = (p, y_0, y_1) and optimise c by
# maximising J(w, c*g_0) over c > 0 via 1-D exact optimisation.
ebproj_update_c = function(fit) {
  x = ebproj_calc_x(fit)
  obj_c = function(log_c) {
    cc = exp(log_c)
    sum(log_Z_binary(x, cc * fit$y_0, cc * fit$y_1, fit$p, fit$sigma2))
  }
  res   = optimize(obj_c, c(-5, 5), maximum = TRUE)
  fit$c = exp(res$maximum)
  return(fit)
}

# Constrained g_0 update: fix c, optimise p with y0, y1 determined by
# mean-0, unit-variance constraint: y1 = sqrt((1-p)/p), y0 = -sqrt(p/(1-p)).
ebproj_update_g0 = function(fit) {
  x = ebproj_calc_x(fit)
  obj_p = function(logit_p) {
    pp  = 1 / (1 + exp(-logit_p))
    y1  =  sqrt((1 - pp) / pp)
    y0  = -sqrt(pp / (1 - pp))
    sum(log_Z_binary(x, fit$c * y0, fit$c * y1, pp, fit$sigma2))
  }
  res      = optimize(obj_p, c(-5, 5), maximum = TRUE)
  pp       = 1 / (1 + exp(-res$maximum))
  fit$p    = pp
  fit$y_1  =  sqrt((1 - pp) / pp)
  fit$y_0  = -sqrt(pp / (1 - pp))
  return(fit)
}

# update tau using rho = vbar' P vbar / U_q, tau = 1/(1-rho); refresh sigma2
ebproj_update_tau = function(fit) {
  ey0  = fit$c * fit$y_0;  ey1 = fit$c * fit$y_1
  x    = ebproj_calc_x(fit)
  pi1  = post_prob1(x, ey0, ey1, fit$p, fit$sigma2)
  vbar = post_mean_binary(x, ey0, ey1, fit$p, fit$sigma2)

  # P = S/n = U D U'/n, so vbar' P vbar = (1/n) (U'vbar)' D (U'vbar)
  Uv         = as.vector(t(fit$U) %*% vbar)
  Uq         = sum((1 - pi1) * ey0^2 + pi1 * ey1^2)   # E_q[v'v] second moment
  rho        = sum(fit$d * Uv^2) / (fit$n * Uq)
  fit$tau    = 1 / (1 - rho)
  fit$sigma2 = fit$n / (fit$nu * fit$tau)
  return(fit)
}

Main fitting loop

# Updates tau, g, w in that order.
# g_update: "g"    = full EM update of (p, y_0, y_1), reset c = 1
#           "c"    = update scale c only, fix g_0
#           "g0"   = update p under mean-0, unit-var constraint, fix c
#           "none" = freeze g entirely
ebproj_fit = function(fit, hessian = "ica",
                      fix_w = FALSE, g_update = "g", fix_tau = FALSE,
                      max_iter = 200, tol = 1e-6, verbose = FALSE) {
  if (!fix_tau && fix_w)
    warning("updating tau without updating w: the objective may not increase")
  fit$objective = ebproj_objective(fit)

  for (iter in seq_len(max_iter)) {
    if (!fix_tau) fit = ebproj_update_tau(fit)

    fit = switch(g_update,
      g    = ebproj_update_g(fit),
      c    = ebproj_update_c(fit),
      g0   = ebproj_update_g0(fit),
      none = fit,
      stop("g_update must be one of: 'g', 'c', 'g0', 'none'")
    )

    if (!fix_w) fit = ebproj_update_w(fit, hessian = hessian)
    
    obj_new = ebproj_objective(fit)

    if (verbose)
      cat(sprintf("iter %3d  obj = %.6f  tau = %.2f  p = %.3f  c = %.4f y0 = %.2f  y1 = %.2f\n",
                  iter, obj_new, fit$tau, fit$p, fit$c, fit$y_0, fit$y_1))

    if (!is.null(fit$objective) && abs(obj_new - fit$objective) < tol) {
      fit$objective = obj_new
      break
    }
    fit$objective = obj_new
  }
  fit$iter = iter
  if(iter==max_iter){
    warning("Warning: iterations reached max_iter iterations")
  }
  return(fit)
}

# Runs fastica from n_starts different starting points and assesses correlation with the true source s_true for each one
run_fastica_seeds = function(U, s_true, b, n_starts = 100) {
  cors = numeric(n_starts)
  for (i in seq_len(n_starts)) {
    fit = ebproj_init_fastica(U, b = b)
    x_hat = ebproj_calc_x(fit)
    cors[i] = abs(cor(x_hat, s_true))
  }
  cors
}

run_ebproj_golden_seeds = function(U, s_true, b, n_starts = 100) {
  c_golden = 2 / (1 + sqrt(5))
  cors = numeric(n_starts)
  for (i in seq_len(n_starts)) {
    fit.init = ebproj_init(U, b = b, c = c_golden, sigma2 = c_golden)
    fit = suppressWarnings(ebproj_fit(fit.init, fix_tau = TRUE, g_update = "g0"))
    x_hat = ebproj_calc_x(fit)
    cors[i] = abs(cor(x_hat, s_true))
  }
  cors
}


run_experiment = function(p_true, n_data_seeds = 10, n_starts = 100,
                          cor_thresh = 0.9, master_seed = 42,
                          n = 200, p_dim = 1000, k = 10) {
  set.seed(master_seed)
  data_seeds = sample.int(1e6, n_data_seeds)
  results = vector("list", n_data_seeds)
  for (di in seq_len(n_data_seeds)) {
    set.seed(data_seeds[di])
    s_raw  = as.numeric(runif(n) < p_true)
    s_true = (s_raw - p_true) / sqrt(p_true * (1 - p_true))
    a = rnorm(p_dim); a = a / sqrt(sum(a^2))
    X = outer(a, s_true) + matrix(rnorm(p_dim * n, 0, 0.1), p_dim, n)
    Xcenter = X - rowMeans(X)
    U = svd(Xcenter, nu = 0, nv = k)$v

    cors_ica    = suppressWarnings(run_fastica_seeds(U, s_true, b = k/2, n_starts = n_starts))
    cors_golden = run_ebproj_golden_seeds(U, s_true, b = k/2, n_starts = n_starts)
    results[[di]] = list(data_seed   = data_seeds[di],
                         frac_ica    = mean(cors_ica    > cor_thresh),
                         frac_golden = mean(cors_golden > cor_thresh))
    cat(sprintf("data seed %7d: fastica %.2f  golden %.2f\n",
                data_seeds[di], results[[di]]$frac_ica, results[[di]]$frac_golden))
  }
  results
}

make_table = function(results, n_starts, cor_thresh) {
  frac_ica         = sapply(results, `[[`, "frac_ica")
  frac_golden      = sapply(results, `[[`, "frac_golden")
  has_goldenplus   = !is.null(results[[1]]$frac_goldenplus)
  tab = data.frame(
    seed         = sapply(results, `[[`, "data_seed"),
    frac_fastica = round(frac_ica, 2),
    frac_golden  = round(frac_golden, 2)
  )
  mean_row = data.frame(seed = NA,
                        frac_fastica = round(mean(frac_ica), 2),
                        frac_golden  = round(mean(frac_golden), 2))
  col_names = c("data seed", "frac found (fastICA)", "frac found (golden)")
  if (has_goldenplus) {
    frac_goldenplus      = sapply(results, `[[`, "frac_goldenplus")

    mean_row$frac_goldenplus = round(mean(frac_goldenplus), 2)
    col_names = c(col_names, "frac found (golden+)")
  }
  tab = rbind(tab, mean_row)
  rownames(tab) = c(paste0("seed ", seq_len(nrow(tab) - 1)), "mean")
  knitr::kable(tab, col.names = col_names,
               caption = sprintf("Fraction of starts with |cor| > %.1f (%d starts each)",
                                 cor_thresh, n_starts))

Test 3: Nine-group data

Here the data come from a 9-group factor model: \(K=9\) binary loading vectors (each with 20 non-zero entries among \(n=100\) observations), a random \(p \times K\) factor matrix \(F\), and small Gaussian noise. The true sources are the columns of \(L\).

For each random start we estimate a single IC and ask whether it is highly correlated (|cor| > 0.9) with any of the \(K=9\) true sources. We whiten to \(k=20\) PCs as in Test 2.

run_fastica_seeds_multi = function(U, L_true, b, n_starts = 100) {
  cors = numeric(n_starts)
  objs = numeric(n_starts)
  for (i in seq_len(n_starts)) {
    fit      = ebproj_init_fastica(U, b = b)
    x_hat    = ebproj_calc_x(fit)
    cors[i]  = max(abs(cor(x_hat, L_true)))
    objs[i]  = ebproj_objective(fit)
  }
  list(cors = cors, objs = objs)
}

run_ebproj_golden_seeds_multi = function(U, L_true, b, n_starts = 100) {
  c_golden = 2 / (1 + sqrt(5))
  cors = numeric(n_starts)
  objs = numeric(n_starts)
  for (i in seq_len(n_starts)) {
    fit.init = ebproj_init(U, b = b, c = c_golden, sigma2 = c_golden)
    fit      = suppressWarnings(ebproj_fit(fit.init, fix_tau = TRUE, g_update = "g0"))
    x_hat    = ebproj_calc_x(fit)
    cors[i]  = max(abs(cor(x_hat, L_true)))
    objs[i]  = fit$objective
  }
  list(cors = cors, objs = objs)
}

run_ebproj_goldenplus_seeds_multi = function(U, L_true, b, n_starts = 100) {
  c_golden = 2 / (1 + sqrt(5))
  cors = numeric(n_starts)
  objs = numeric(n_starts)
  for (i in seq_len(n_starts)) {
    fit.init    = ebproj_init(U, b = b, c = c_golden, sigma2 = c_golden)
    fit_golden  = suppressWarnings(ebproj_fit(fit.init, fix_tau = TRUE, g_update = "g0"))
    fit         = ebproj_fit(fit_golden)   # full updates from golden solution
    x_hat       = ebproj_calc_x(fit)
    cors[i]     = max(abs(cor(x_hat, L_true)))
    objs[i]     = fit$objective
  }
  list(cors = cors, objs = objs)
}

run_experiment_test3 = function(n_data_seeds = 10, n_starts = 100,
                                cor_thresh = 0.9, master_seed = 42,
                                n = 100, p_dim = 1000, K = 9, k = 20) {
  set.seed(master_seed)
  data_seeds = sample.int(1e6, n_data_seeds)
  results = vector("list", n_data_seeds)
  for (di in seq_len(n_data_seeds)) {
    set.seed(data_seeds[di])
    L  = matrix(0, nrow = n, ncol = K)
    for (i in seq_len(K)) L[sample(n, 20), i] = 1
    FF = matrix(rnorm(p_dim * K), nrow = p_dim)
    X  = t(L %*% t(FF) + matrix(rnorm(n * p_dim, 0, 0.1), nrow = n))
    Xcenter = X - rowMeans(X)
    U = svd(Xcenter, nu = 0, nv = k)$v

    out_ica        = suppressWarnings(run_fastica_seeds_multi(U, L, b = k/2, n_starts = n_starts))
    out_golden     = run_ebproj_golden_seeds_multi(U, L, b = k/2, n_starts = n_starts)
    out_goldenplus = run_ebproj_goldenplus_seeds_multi(U, L, b = k/2, n_starts = n_starts)
    results[[di]] = list(data_seed         = data_seeds[di],
                         cors_ica          = out_ica$cors,
                         objs_ica          = out_ica$objs,
                         cors_golden       = out_golden$cors,
                         objs_golden       = out_golden$objs,
                         cors_goldenplus   = out_goldenplus$cors,
                         objs_goldenplus   = out_goldenplus$objs,
                         frac_ica          = mean(out_ica$cors        > cor_thresh),
                         frac_golden       = mean(out_golden$cors     > cor_thresh),
                         frac_goldenplus   = mean(out_goldenplus$cors > cor_thresh))
    cat(sprintf("data seed %7d: fastica %.2f  golden %.2f  golden+ %.2f\n",
                data_seeds[di], results[[di]]$frac_ica,
                results[[di]]$frac_golden, results[[di]]$frac_goldenplus))
  }
  results
}

plot_objectives_t3 = function(results, cor_thresh = 0.9,
                              method = c("ica", "golden", "goldenplus")) {
  method   = match.arg(method)
  n_seeds  = length(results)
  par(mfrow = c(2, ceiling(n_seeds / 2)), mar = c(3, 3, 2, 1))
  for (di in seq_len(n_seeds)) {
    cors = results[[di]][[paste0("cors_", method)]]
    objs = results[[di]][[paste0("objs_", method)]]
    ord  = order(objs, decreasing = TRUE)
    cols = ifelse(cors[ord] > cor_thresh, "red", "blue")
    plot(objs[ord], col = cols, pch = 16, cex = 0.7,
         xlab = "rank (by objective)", ylab = "objective",
         main = sprintf("seed %d", results[[di]]$data_seed))
  }
  par(mfrow = c(1, 1))
}
